📦 אובייקטים — מבנה נתונים

מה זה אובייקט?

אובייקט הוא אוסף של זוגות key: value — כמו כרטיס זהות!

let person = { name: "ישראל כהן", // key: value age: 28, city: "ירושלים", isStudent: true }
name:
age:
city:
isStudent:

גישה לנכסים — שתי דרכים

// דרך 1 — נקודה (dot notation) — הכי נפוץ person.name // "ישראל כהן" // דרך 2 — סוגריים מרובעות (bracket notation) person["age"] // 28 // מה ההבדל? // סוגריים נדרשות כשה-key הוא משתנה: let field = "city" person[field] // "ירושלים"

שינוי, הוספה ומחיקת נכסים

person.age = 29 // שינוי person.email = "x@y.com" // הוספה delete person.isStudent // מחיקה
