🏗️ יצירת אלמנטים דינמית

שיטה 1: innerHTML (פשוטה)

let container = document.getElementById("container1") container.innerHTML += `<div class="item">פריט חדש!</div>`

שיטה 2: createElement (מומלץ ובטוח יותר)

let div = document.createElement("div") // צור אלמנט div.className = "dynamic-item" // הגדר class div.textContent = "תוכן בטוח" // ← textContent ולא innerHTML! container.appendChild(div) // הוסף לדף

בניית כרטיסיות מאובייקטים

let products = [ { name: "עפרון", price: 5 }, { name: "מחברת", price: 15 } ] products.forEach(product => { let card = document.createElement("div") card.className = "card" card.innerHTML = ` <strong>${product.name}</strong> <span>₪${product.price}</span> ` container.appendChild(card) })
