⛔ break ו-⏭️ continue

break — עצור את הלולאה לגמרי

// מצא את המספר הראשון שמתחלק ב-7 for (let i = 1; i <= 100; i++) { if (i % 7 === 0) { console.log(`מצאתי! ${i}`) break // ← עוצר כאן, לא ממשיך } }

continue — דלג לאיטרציה הבאה

// הדפס רק מספרים שאינם מתחלקים ב-3 for (let i = 1; i <= 15; i++) { if (i % 3 === 0) { continue // ← דלג, חזור לתחילת הלולאה } console.log(i) }

לולאת while — שימוש מציאותי

// חפש מספר שנוחש על ידי המשתמש let secret = 42 let guess = 0 let attempts = 0 while (guess !== secret) { guess = Number(prompt("נחש!")) attempts++ if (attempts > 5) { alert("יותר מדי ניסיונות!") break } }
