🗺️ map ו-🔍 filter — כלי העל!

map — עושה משהו לכל איבר, מחזיר מערך חדש

let prices = [100, 200, 300, 400, 500] // הכפל כל מחיר ב-1.17 (מע"מ) let withTax = prices.map(price => price * 1.17) // [117, 234, 351, 468, 585]

filter — מחזיר רק איברים שעוברים תנאי

let grades = [55, 72, 88, 45, 95, 60, 78, 33] // רק ציונים 70 ומעלה let passing = grades.filter(grade => grade >= 70) // [72, 88, 95, 78]
